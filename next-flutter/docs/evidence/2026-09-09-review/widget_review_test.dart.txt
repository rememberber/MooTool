import 'dart:io';
import 'dart:ui' as ui;
import 'package:flutter/material.dart';
import 'package:flutter/rendering.dart';
import 'package:flutter/services.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:mootool_next_flutter/main.dart';
import 'package:mootool_next_flutter/app/app_controller.dart';
import 'package:mootool_next_flutter/app/app_paths.dart';
import 'package:mootool_next_flutter/app/tool_registry.dart';
import 'package:mootool_next_flutter/core/desktop/desktop_host.dart';
import 'package:mootool_next_flutter/core/editor/editor_document.dart';
import 'package:mootool_next_flutter/design/widgets.dart';
import 'package:mootool_next_flutter/design/theme.dart';

void main() {
  for (final width in [1440.0, 1080.0]) {
    testWidgets('all tool pages fit ${width.toInt()} x 900', (tester) async {
      tester.view.physicalSize = Size(width, 900);
      tester.view.devicePixelRatio = 1;
      addTearDown(tester.view.resetPhysicalSize);
      addTearDown(tester.view.resetDevicePixelRatio);
      final c = await tester.runAsync(() async {
        final root = await Directory.systemTemp.createTemp('mootool-widget-review-');
        final controller = AppController(AppPaths(root), desktopHost: MemoryDesktopHost());
        await controller.load();
        for (final family in ['Roboto','Ahem','monospace']) {
          final font = FontLoader(family);
          font.addFont(File('/System/Library/Fonts/Supplemental/Arial Unicode.ttf').readAsBytes().then((b) => b.buffer.asByteData()));
          await font.load();
        }
        final icons = FontLoader('MaterialIcons');
        icons.addFont(File('/Users/zhoubo/sdk/flutter/bin/cache/artifacts/material_fonts/MaterialIcons-Regular.otf').readAsBytes().then((b) => b.buffer.asByteData()));
        await icons.load();
        addTearDown(() async {
          controller.dispose();
          if (root.existsSync()) await root.delete(recursive:true);
        });
        return controller;
      });
      final failures = <String>[];
      final boundary = GlobalKey();
      final originalHandler = FlutterError.onError;
      var currentId = '';
      FlutterError.onError = (details) => failures.add('$currentId @ ${width.toInt()}: ${details.exceptionAsString()}');
      addTearDown(() => FlutterError.onError = originalHandler);
      for (final id in [...toolRegistry.map((t) => t.id), 'settings']) {
        currentId = id;
        c!.activeToolId = id == 'settings' ? 'mootool' : id;
        if (id == 'settings') { c.openSettings(); } else { c.closeOverlays(); }
        await tester.pumpWidget(RepaintBoundary(key: boundary, child: MooToolApp(controller:c)));
        await tester.pump(const Duration(milliseconds:50));
        Object? error;
        while ((error = tester.takeException()) != null) {
          failures.add('$id @ ${width.toInt()}: $error');
        }
        if (width == 1440 && ['json', 'quickNote', 'http', 'calculator', 'settings'].contains(id)) {
          final node = boundary.currentContext!.findRenderObject()! as RenderRepaintBoundary;
          await tester.runAsync(() async {
            final image = await node.toImage();
            final data = await image.toByteData(format: ui.ImageByteFormat.png);
            await File('/tmp/mootool-flutter-review.6VnMrU/$id.png').writeAsBytes(data!.buffer.asUint8List());
            image.dispose();
          });
        }
      }
      await tester.pumpWidget(const SizedBox.shrink());
      FlutterError.onError = originalHandler;
      expect(failures.toSet(), isEmpty, reason: failures.toSet().join('\n'));
    });
  }
  testWidgets('R12 selection-only find result reaches visible editor', (tester) async {
    final doc = EditorDocument(id:'probe', text:'alpha beta gamma');
    Widget view() => MaterialApp(theme:buildTheme(brightness:Brightness.light,uiFontSize:13), home:Scaffold(body:MooCodeEditor(document:doc,onChanged:(_){})));
    await tester.pumpWidget(view());
    doc.restoreView(start:6,end:10);
    await tester.pumpWidget(view());
    final field = tester.widget<TextField>(find.byType(TextField));
    expect(field.controller!.selection, const TextSelection(baseOffset:6,extentOffset:10));
  });
}
